package admission

import (
	"bytes"
	"context"
	"encoding/json"
	"io"
	"net/http"
	"os"
	"path/filepath"
	"strings"
	"testing"
	"time"
	"vansail-ip-atlas/internal/acquire"
	"vansail-ip-atlas/internal/intelligence"
)

func TestOriginalEditionAdmission(t *testing.T) {
	ctx, cancel := context.WithTimeout(context.Background(), 4*time.Minute)
	defer cancel()
	now := time.Now().UTC().Truncate(time.Second)
	root, err := filepath.Abs("../..")
	if err != nil {
		t.Fatal("workspace")
	}
	file, err := os.Open(filepath.Join(root, "configs/sources.yaml"))
	if err != nil {
		t.Fatal("catalog")
	}
	all, err := DecodeCatalog(file)
	file.Close()
	if err != nil {
		t.Fatal("catalog")
	}
	var source Source
	for _, s := range all {
		if s.ID == intelligence.AppleRelaySourceID {
			source = s
		}
	}
	engine, err := New(root)
	if err != nil {
		t.Fatal("source engine")
	}
	defer engine.Close()

	query := appleWitnessAPI + "?path=" + appleWitnessFile + "&sha=master&per_page=1"
	req, _ := http.NewRequestWithContext(ctx, "GET", query, nil)
	req.Header.Set("User-Agent", "VanSail-IP-Atlas/3")
	probeResponse, httpErr := http.DefaultClient.Do(req)
	if httpErr == nil {
		t.Logf("witness_http: status=%d rate_exhausted=%t", probeResponse.StatusCode, probeResponse.Header.Get("X-RateLimit-Remaining") == "0")
		probeResponse.Body.Close()
	}
	dir := t.TempDir()
	path := filepath.Join(dir, "original")
	current, parent, currentErr := engine.appleWitness(ctx, dir, "current", "master")
	t.Logf("current_evidence: passed=%t prefixes=%d", currentErr == nil, current.Prefixes)

	if currentErr != nil {
		b, er := os.ReadFile(filepath.Join(dir, "apple-current-commit"))
		var rows []appleCommit
		validObj := len(b) > 0 && uniqueJSON(b) == nil && json.Unmarshal(b, &rows) == nil && len(rows) == 1
		t.Logf("witness_metadata: downloaded=%t parsed=%t", er == nil, validObj)
		if validObj {
			m := rows[0]
			t.Logf("witness_commit: identity=%t parent=%t same_dates=%t edition_previously_checked=%t", commitSHA(m.SHA) && m.URL == appleWitnessAPI+"/"+m.SHA, len(m.Parents) == 1, m.Commit.Author.Date == m.Commit.Committer.Date, m.SHA == "120b5e4958c8af42d837583978b9a53cda887475")
			data, er := os.ReadFile(filepath.Join(dir, "apple-current-prefixes"))
			t.Logf("witness_prefix_file: downloaded=%t bytes=%d", er == nil, len(data))
			if er == nil {
				prefixes, _, parseErr := readApplePrefixes(ctx, bytes.NewReader(data), true, intelligence.AppleRelayMaxPrefixes, nil)
				t.Logf("witness_prefix_shape: complete=%t count=%d source_declared=%t", parseErr == nil, len(prefixes), bytes.Contains(data, []byte("# source: "+acquire.AppleRelayURL+"\n")))
			} else {
				alt, er := http.DefaultClient.Get(appleWitnessRaw + m.SHA + "/" + appleWitnessFile)
				if er == nil {
					n, err := io.Copy(io.Discard, io.LimitReader(alt.Body, 32<<20))
					alt.Body.Close()
					t.Logf("immutable_file_http: status=%d bytes=%d complete=%t gzip=%t", alt.StatusCode, n, err == nil, strings.EqualFold(alt.Header.Get("Content-Encoding"), "gzip"))
				}
			}
		}
	}
	if currentErr == nil {
		earlier, _, er := engine.appleWitness(ctx, dir, "previous", parent)
		t.Logf("preceding_evidence: passed=%t prefixes=%d changed=%t", er == nil, earlier.Prefixes, er == nil && earlier.PrefixHash != current.PrefixHash)
	}
	v, err := engine.fetch.Fetch(ctx, acquire.AppleRelayURL, path, source.MaxCompressedBytes)
	t.Logf("original_download: passed=%t bytes=%d", err == nil, v.Bytes)
	if err == nil {
		original, err := os.Open(path)
		if err != nil {
			t.Fatal("local original")
		}
		prefixes, _, err := readApplePrefixes(ctx, original, false, intelligence.AppleRelayMaxPrefixes, nil)
		original.Close()
		t.Logf("original_projection: passed=%t prefix_count=%d witness_equal=%t", err == nil, len(prefixes), err == nil && currentErr == nil && prefixes.Hash() == current.PrefixHash)
		verifier := engine.fetch.(*acquire.Fetcher)
		_, clockErr := verifier.VerifyAppleRelayClock(ctx, v.ETag, v.Bytes)
		t.Logf("operator_clock: passed=%t", clockErr == nil)
	}
	binding := [32]byte{1}
	res, err := engine.IngestPartitioned(ctx, source, now, binding)
	if err != nil || res.Snapshot == nil {
		t.Fatal("original_admission: failed")
	}
	plan, err := intelligence.ReconcilePartitioned(ctx, root, intelligence.BuildState{}, []intelligence.SourcePolicy{res.Policy}, []intelligence.SourceSnapshot{*res.Snapshot}, now, binding)
	if err != nil {
		t.Fatal("original_reconciliation: failed")
	}
	_, stats, err := RevalidateStoredSources(ctx, root, []Source{source}, plan.State, now, binding)
	if err != nil || len(stats) != 1 {
		t.Fatal("original_revalidation: failed")
	}
	t.Logf("original_admission: passed; revalidated_prefixes=%d", stats[0].Records)
}
